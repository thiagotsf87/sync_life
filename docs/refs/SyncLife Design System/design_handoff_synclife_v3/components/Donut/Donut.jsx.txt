import React from 'react';

/**
 * Donut — a conic-gradient share chart with a centred value and a legend of
 * labelled slices. `share` values are percentages (they map to degrees, so
 * they should sum to ~100).
 */
export function Donut({ title, center, centerSub, slices = [] }) {
  let acc = 0;
  const stops = slices.map((s) => {
    const from = acc; acc += s.share * 3.6;
    return `${s.color} ${from}deg ${acc}deg`;
  }).join(', ');
  return (
    <div style={{
      background: 'var(--sl-bg-surface, #11161D)', border: '1px solid var(--sl-line-subtle, rgba(255,255,255,0.07))',
      borderRadius: 16, padding: 22, display: 'flex', flexDirection: 'column', gap: 16,
      fontFamily: 'var(--sl-font-body, sans-serif)',
    }}>
      {title && <h3 style={{ fontFamily: 'var(--sl-font-display, sans-serif)', fontSize: 16, fontWeight: 600, margin: 0, color: 'var(--sl-fg-primary, #E7ECF1)', letterSpacing: '-0.01em' }}>{title}</h3>}
      <div style={{ display: 'flex', alignItems: 'center', gap: 22 }}>
        <div style={{ width: 132, height: 132, borderRadius: 999, background: `conic-gradient(${stops})`, flexShrink: 0, display: 'flex', alignItems: 'center', justifyContent: 'center' }}>
          <div style={{ width: 78, height: 78, borderRadius: 999, background: 'var(--sl-bg-surface, #11161D)', display: 'flex', flexDirection: 'column', alignItems: 'center', justifyContent: 'center' }}>
            <span style={{ fontFamily: 'var(--sl-font-display, sans-serif)', fontWeight: 600, fontSize: 18, color: 'var(--sl-fg-primary, #E7ECF1)', lineHeight: 1 }}>{center}</span>
            {centerSub && <span style={{ fontSize: 9.5, color: 'var(--sl-fg-muted, #4A535F)', marginTop: 3 }}>{centerSub}</span>}
          </div>
        </div>
        <div style={{ flex: 1, display: 'flex', flexDirection: 'column', gap: 8 }}>
          {slices.map((s, i) => (
            <div key={i} style={{ display: 'flex', alignItems: 'center', gap: 9, fontSize: 12.5 }}>
              <span style={{ width: 9, height: 9, borderRadius: 3, background: s.color }}/>
              <span style={{ flex: 1, color: 'var(--sl-fg-secondary, #A9B2BD)' }}>{s.label}</span>
              <span style={{ fontFamily: 'var(--sl-font-mono, monospace)', color: 'var(--sl-fg-primary, #E7ECF1)', fontWeight: 600 }}>{s.share}%</span>
            </div>
          ))}
        </div>
      </div>
    </div>
  );
}
