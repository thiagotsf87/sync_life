import React from 'react';

const Check = ({ size = 9 }) => (
  <svg width={size} height={size} viewBox="0 0 24 24" fill="none" stroke="#0B0F14"
    strokeWidth="3.5" strokeLinecap="round" strokeLinejoin="round" style={{ display: 'block' }}>
    <polyline points="20 6 9 17 4 12"/>
  </svg>
);

/**
 * Timeline — a vertical milestone / roadmap list. Each item has a `when`
 * column, a connector node (filled when `done`, ringed + "AGORA" when
 * `current`), and a title + meta. The connector line runs between nodes.
 */
export function Timeline({ title, items = [] }) {
  return (
    <div style={{
      background: 'var(--sl-bg-surface, #11161D)', border: '1px solid var(--sl-line-subtle, rgba(255,255,255,0.07))',
      borderRadius: 16, padding: 22, fontFamily: 'var(--sl-font-body, sans-serif)',
    }}>
      {title && <h3 style={{ fontFamily: 'var(--sl-font-display, sans-serif)', fontSize: 16, fontWeight: 600, margin: '0 0 18px', color: 'var(--sl-fg-primary, #E7ECF1)', letterSpacing: '-0.01em' }}>{title}</h3>}
      <div style={{ display: 'flex', flexDirection: 'column' }}>
        {items.map((it, i) => {
          const last = i === items.length - 1;
          const accent = 'var(--sl-accent, #34D399)';
          return (
            <div key={i} style={{ display: 'grid', gridTemplateColumns: '90px 32px 1fr', gap: 12, alignItems: 'stretch' }}>
              <div style={{ textAlign: 'right', paddingTop: 2 }}>
                <span style={{ fontFamily: 'var(--sl-font-mono, monospace)', fontSize: 11.5, color: it.current ? accent : 'var(--sl-fg-tertiary, #6B7482)', fontWeight: it.current ? 700 : 500 }}>{it.when}</span>
              </div>
              <div style={{ display: 'flex', flexDirection: 'column', alignItems: 'center' }}>
                <span style={{
                  width: 16, height: 16, borderRadius: 999, flexShrink: 0,
                  background: it.done ? accent : it.current ? 'var(--sl-accent-soft, rgba(52,211,153,0.14))' : 'var(--sl-bg-elevated, rgba(255,255,255,0.05))',
                  border: `2px solid ${it.done ? accent : it.current ? accent : (it.color || 'var(--sl-line-strong, rgba(255,255,255,0.16))')}`,
                  display: 'inline-flex', alignItems: 'center', justifyContent: 'center',
                }}>{it.done && <Check size={9}/>}</span>
                {!last && <span style={{ flex: 1, width: 2, background: it.done ? accent : 'var(--sl-line-subtle, rgba(255,255,255,0.07))', minHeight: 22 }}/>}
              </div>
              <div style={{ paddingBottom: last ? 0 : 20 }}>
                <div style={{ fontSize: 13.5, fontWeight: 600, color: 'var(--sl-fg-primary, #E7ECF1)' }}>
                  {it.title}
                  {it.current && <span style={{ marginLeft: 8, fontSize: 9.5, fontWeight: 700, color: accent, letterSpacing: '0.06em' }}>AGORA</span>}
                </div>
                {it.meta && <div style={{ fontSize: 12, color: 'var(--sl-fg-tertiary, #6B7482)', marginTop: 2 }}>{it.meta}</div>}
              </div>
            </div>
          );
        })}
      </div>
    </div>
  );
}
